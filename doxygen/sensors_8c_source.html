132can_msg conv2CANVelocityData(bool vehicle_direction, double relative_velocity, double relative_acceleration)
133{
134 //printf("Rel acel: %lf\n", relative_acceleration);
135 can_msg aux = {.identifier = ID_SPEED_S, .dataFrame = BASE_DATA_FRAME};
136
137 // Vehicle direction (forward or reverse) data encapsulation
138 if (vehicle_direction)
139 {
140 aux.dataFrame[2] = 0x01;
141 }
142 else
143 {
144 aux.dataFrame[2] = 0x00;
145 }
146
147 // Speed data ​​encapsulation
148 unsigned int data_speed = relative_velocity / RES_SPEED_S;
149 unsigned char ms_speed, ls_speed;
150 ls_speed = data_speed;
151 ms_speed = data_speed >> 8;
152
153 // Defines most and least significant bytes, according to the DBC specification
154 aux.dataFrame[0] = ls_speed;
155 aux.dataFrame[1] = ms_speed;
156
157 // Acceleration data ​​encapsulation
158 double aux_acel = relative_acceleration;
159 if(aux_acel < 0){
160 aux_acel *= -1;
161 aux.dataFrame[5] = 0x01;
162 } else {
163 aux.dataFrame[5] = 0x00;
164 }
165
166 unsigned int data_acel = ((aux_acel * RES_ACCELERATION_DIV_S) - OFFSET_ACCELERATION_S);
167 unsigned char ms_acel, ls_acel;
168 ls_acel = data_acel;
169 ms_acel = data_acel >> 8;
170
171 // Defines most and least significant bytes, according to the DBC specification
172 aux.dataFrame[3] = ls_acel;
173 aux.dataFrame[4] = ms_acel;
174
175 return aux;
176}
